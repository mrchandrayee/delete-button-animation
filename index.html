app.css

/* THE MOUTH NEVER MOVES. The scale
origin is the body's own TOP EDGE,
dead centre, so the rim stays
pinned while the bin grows DOWN
past the bottom of the button,
and the word has somewhere to fall */
.maw__can {
  transform-origin: var(--pivot-x)
                   var(--pivot-y);
  transform: translateX(var(--shift))
             scale(var(--z));
}

/* the twin is SOLVED from the label,
   never tweened beside it */
.maw__twin {
  --gy: calc(var(--anchor-y)
    + var(--my) - var(--can-y)
    - var(--pivot-y));
  transform: scale(calc(1/var(--z)))
    translate(var(--gx), var(--gy))
    rotate(var(--wrot));
}

app.js

// the bin's width comes from the
// ICON and never changes; the
// word's comes from the TEXT, so
// one fixed scale fits exactly
// ONE label: "Remove" was 104% of
// the bin and hung out of it.
const bin = num('--body-w')
  * num('--z-open');
const room =
  bin * num('--word-fill');

// what must fit is its ROTATED
// extent, not its width
const ext = lw * Math.cos(tip)
  + lh * Math.sin(tip);
const sc = Math.min(
  num('--word-sc-max'), room / ext
);

// and it rests DEAD CENTRE of the
// bin, so it cannot leave sideways
set('--word-x',
  w / 2 - bboxL - ext * sc / 2);
